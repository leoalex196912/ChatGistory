def make_countersink_cone(r_bottom, r_top, height):
    """Proper 90 degree cone for M5 flathead countersink."""
    return Part.makeCone(r_bottom, r_top, height)

def apply_outer_chamfer(shape, size):
    """V2.1-E: Chamfer only outer perimeter vertical edges.
    Uses bounding box proximity check so internal edges (pocket walls,
    shaft bore) are never accidentally chamfered even if geometry changes.
    """
    try:
        bb = shape.BoundBox
        tol = 0.1  # 0.1mm tolerance for 'on the perimeter' test
        edges_to_chamfer = []
        for edge in shape.Edges:
            v1, v2 = edge.Vertexes[0].Point, edge.Vertexes[1].Point
            # Must be a vertical edge (parallel to Z)
            if abs(v1.z - v2.z) < 1.0:
                continue
            # Must lie on the outer perimeter (X or Y at the bounding box limit)
            on_perimeter = (
                abs(v1.x - bb.XMin) < tol or abs(v1.x - bb.XMax) < tol or
                abs(v1.y - bb.YMin) < tol or abs(v1.y - bb.YMax) < tol
            )
            if on_perimeter:
                edges_to_chamfer.append(edge)
        if edges_to_chamfer:
            return shape.makeChamfer(size, edges_to_chamfer)
        return shape
    except Exception as e:
        print(f"  Chamfer skipped: {e}")
        return shape

    ang = math.radians(i * 90.0 + 45.0)
    bx = (TOP_BOLT_PCD / 2.0) * math.cos(ang)
    by = (TOP_BOLT_PCD / 2.0) * math.sin(ang)

    cs.translate(FreeCAD.Vector(bx, by, 0))
    top = top.cut(cs)
